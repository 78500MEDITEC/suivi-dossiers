/** * SUIVI DE DOSSIERS MEDITEC — Relances automatiques + Google Calendar */ // ==================== CONFIGURATION ==================== const CONFIG = { SHEET_NAME: "Dossiers", GERANT_EMAIL: "meditec78@gmail.com", // à remplacer CALENDAR_NAME: "Suivi Dossiers MEDITEC", // calendrier partagé dédié JOURS_AVANT_RELANCE: 2, // délai entre deux relances si non traité RELANCES_AVANT_ALERTE_GERANT: 3, STATUTS_OUVERTS: ["Nouveau", "En cours", "Relance"] }; // ==================== MENU ==================== function onOpen() { SpreadsheetApp.getUi() .createMenu("Suivi Dossiers") .addItem("Initialiser la feuille", "initializeSheet") .addItem("Vérifier les relances maintenant", "checkDossiers") .addItem("Marquer la ligne sélectionnée comme Clôturé", "marquerCommeTraite") .addToUi(); } function initializeSheet() { const sheet = getSheet(); const headers = ["ID_Dossier","Client","Type_Demande","Salarié_Assigné","Email_Salarié", "Date_Création","Date_Relance","Statut","Nb_Relances","Dernière_Relance","Notes","Event_ID", "Client_Email","Client_Téléphone","Event_Link","Suite_Donnée","Canal_Contact","Ref_Orthopmust", "Créé_Par","Modifié_Par","Date_Modification","Nature_Contact","Société","Montant_Avoir","Dossiers_Liés", "Référent_Type","Référent_Nom","Référent_Contact","Établissement"]; if (sheet.getLastRow() === 0) { sheet.appendRow(headers); sheet.setFrozenRows(1); } const salariesSheet = getSalariesSheet(); if (salariesSheet.getLastRow() === 0) { salariesSheet.appendRow(["Nom", "Email"]); salariesSheet.setFrozenRows(1); } ensureCalendarExists(); sheet.getRange("N2:N").setNumberFormat("@"); // colonne Client_Téléphone toujours en texte (garde le 0 initial) SpreadsheetApp.getUi().alert("Feuille initialisée. Vous pouvez ajouter vos dossiers."); } function getSheet() { return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG.SHEET_NAME) || SpreadsheetApp.getActiveSpreadsheet().insertSheet(CONFIG.SHEET_NAME); } function getSalariesSheet() { return SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Salaries") || SpreadsheetApp.getActiveSpreadsheet().insertSheet("Salaries"); } // ==================== CALENDRIER PARTAGÉ ==================== function ensureCalendarExists() { let calendars = CalendarApp.getCalendarsByName(CONFIG.CALENDAR_NAME); if (calendars.length === 0) { CalendarApp.createCalendar(CONFIG.CALENDAR_NAME); } } function getSharedCalendar() { const calendars = CalendarApp.getCalendarsByName(CONFIG.CALENDAR_NAME); return calendars.length > 0 ? calendars[0] : CalendarApp.createCalendar(CONFIG.CALENDAR_NAME); } /** * Construit l'URL Google Calendar d'un événement à partir de son ID * et de l'ID du calendrier (getHtmlLink() n'existe pas sur les événements * créés via Apps Script, il faut reconstruire le lien manuellement). */ function construireLienEvenement(eventId, calendarId) { const eid = Utilities.base64EncodeWebSafe(eventId + " " + calendarId).replace(/=+$/, ""); return "https://calendar.google.com/calendar/event?eid=" + eid; } /** * Crée un événement dans le calendrier partagé pour un dossier, * avec le salarié assigné en invité (il reçoit une invitation Google Calendar). */ function creerEvenementDossier(idDossier, client, typeDemande, salarie, emailSalarie, dateRelance) { const calendar = getSharedCalendar(); const titre = `[${typeDemande}] ${client} — ${salarie}`; const event = calendar.createAllDayEvent(titre, dateRelance, { description: `Dossier ${idDossier}\nClient : ${client}\nAssigné à : ${salarie}\nÀ traiter avant le ${Utilities.formatDate(dateRelance, Session.getScriptTimeZone(), "dd/MM/yyyy")}`, guests: emailSalarie, sendInvites: true }); event.addPopupReminder(60); // rappel 1h avant, ajustable return { id: event.getId(), link: construireLienEvenement(event.getId(), calendar.getId()) }; } /** * Crée un événement à horaire précis (vrai rendez-vous), plutôt qu'un simple * rappel sur toute la journée. Durée par défaut : 1h. */ function creerEvenementRendezVous(idDossier, client, typeDemande, salarie, emailSalarie, dateHeureDebut) { const calendar = getSharedCalendar(); const titre = `[RDV ${typeDemande}] ${client} — ${salarie}`; const dateHeureFin = new Date(dateHeureDebut.getTime() + 60 * 60 * 1000); // +1h const event = calendar.createEvent(titre, dateHeureDebut, dateHeureFin, { description: `Rendez-vous — Dossier ${idDossier}\nClient : ${client}\nAssigné à : ${salarie}`, guests: emailSalarie, sendInvites: true }); event.addPopupReminder(60); return { id: event.getId(), link: construireLienEvenement(event.getId(), calendar.getId()) }; } // ==================== VÉRIFICATION QUOTIDIENNE (déclencheur automatique) ==================== function checkDossiers() { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); const today = new Date(); today.setHours(0,0,0,0); for (let i = 1; i < data.length; i++) { const row = data[i]; const [idDossier, client, typeDemande, salarie, emailSalarie, dateCreation, dateRelance, statut, nbRelances, derniereRelance, notes] = row; if (!CONFIG.STATUTS_OUVERTS.includes(statut)) continue; // dossier clôturé, on ignore if (!dateRelance) continue; const relanceDate = new Date(dateRelance); relanceDate.setHours(0,0,0,0); if (today >= relanceDate) { envoyerRelance(sheet, i + 1, idDossier, client, salarie, emailSalarie, nbRelances || 0, notes); } } } function envoyerRelance(sheet, ligneExcel, idDossier, client, salarie, emailSalarie, nbRelances, notes) { const nouvelleValeurNbRelances = nbRelances + 1; // Email au salarié assigné MailApp.sendEmail({ to: emailSalarie, subject: `[Relance] Dossier ${client} à traiter`, body: `Bonjour ${salarie},\n\nLe dossier "${client}" (${idDossier}) n'est pas encore clôturé.\n\nNotes : ${notes || "—"}\n\nMerci de le traiter et de mettre à jour son statut dans le Sheet "Suivi Dossiers MEDITEC".\n\n— Relance automatique` }); // Escalade au gérant après X relances if (nouvelleValeurNbRelances >= CONFIG.RELANCES_AVANT_ALERTE_GERANT) { MailApp.sendEmail({ to: CONFIG.GERANT_EMAIL, subject: `[ALERTE] Dossier ${client} non traité après ${nouvelleValeurNbRelances} relances`, body: `Le dossier "${client}" assigné à ${salarie} n'a toujours pas été clôturé après ${nouvelleValeurNbRelances} relances.\n\nID dossier : ${idDossier}\nNotes : ${notes || "—"}` }); } // Mise à jour de la feuille : statut, compteur, date, et report de la prochaine relance const prochaineRelance = new Date(); prochaineRelance.setDate(prochaineRelance.getDate() + CONFIG.JOURS_AVANT_RELANCE); sheet.getRange(ligneExcel, 8).setValue("Relance"); // H: Statut sheet.getRange(ligneExcel, 9).setValue(nouvelleValeurNbRelances); // I: Nb_Relances sheet.getRange(ligneExcel, 10).setValue(new Date()); // J: Dernière_Relance sheet.getRange(ligneExcel, 7).setValue(prochaineRelance); // G: Date_Relance (report) } // ==================== MARQUER COMME TRAITÉ ==================== function marquerCommeTraite() { const sheet = getSheet(); const range = sheet.getActiveRange(); const ligne = range.getRow(); if (ligne === 1) { SpreadsheetApp.getUi().alert("Sélectionnez une ligne de dossier, pas l'en-tête."); return; } sheet.getRange(ligne, 8).setValue("Clôturé"); // colonne H: Statut SpreadsheetApp.getUi().alert("Dossier marqué comme Clôturé."); } // ==================== CRÉATION AUTOMATIQUE DE L'ÉVÉNEMENT (dès qu'une ligne est complétée) ==================== /** * Déclencheur INSTALLABLE (pas un simple onEdit) : à installer manuellement * via le menu Déclencheurs (voir Étape 4 bis). Dès qu'un salarié remplit * Email_Salarié (colonne E) ou Date_Relance (colonne G) sur une ligne, * l'événement Calendar est créé automatiquement, sans aucun clic. */ function onDossierEdit(e) { const sheet = e.range.getSheet(); if (sheet.getName() !== CONFIG.SHEET_NAME) return; const ligne = e.range.getRow(); if (ligne === 1) return; // en-tête const colonneModifiee = e.range.getColumn(); if (colonneModifiee !== 5 && colonneModifiee !== 7) return; // seulement E (Email) ou G (Date_Relance) const data = sheet.getRange(ligne, 1, 1, 12).getValues()[0]; const [idDossier, client, typeDemande, salarie, emailSalarie, dateCreation, dateRelance, statut, nbRelances, derniereRelance, notes, eventId] = data; if (!emailSalarie || !dateRelance) return; // pas encore assez d'infos sur la ligne if (eventId) return; // événement déjà créé pour cette ligne, on ne duplique pas const resultat = creerEvenementDossier(idDossier, client, typeDemande, salarie, emailSalarie, new Date(dateRelance)); sheet.getRange(ligne, 12).setValue(resultat.id); // colonne L : Event_ID sheet.getRange(ligne, 15).setValue(resultat.link); // colonne O : Event_Link } // ==================== LIAISON AVEC LE CRM EXISTANT ==================== /** * Fonction à appeler depuis votre CRM existant (module d'intake de contact) * pour créer automatiquement une ligne "Dossiers" dès qu'un contact arrive * (téléphone, mail, boutique, démarchage) — sans ressaisie manuelle. * * Si le CRM tourne dans le MÊME projet Apps Script que ce script : * appelez directement ajouterDossierDepuisCRM(...) depuis le code du CRM. * * Si le CRM est un projet Apps Script SÉPARÉ : * remplacez getSheet() ci-dessous par : * SpreadsheetApp.openById("ID_DU_SHEET_SUIVI_DOSSIERS").getSheetByName(CONFIG.SHEET_NAME) * (l'ID du Sheet se trouve dans son URL, entre /d/ et /edit) */ function ajouterDossierDepuisCRM(client, typeDemande, salarieAssigne, emailSalarie, notes, joursAvantRelance, clientEmail, clientTelephone, rdvDateTime, canalContact, refOrthopmust, creePar, natureContact, societe, montantAvoir, dossierLie, referentType, referentNom, referentContact, etablissement) { const sheet = getSheet(); const id = "DOS-" + new Date().getTime(); const dateCreation = new Date(); let dateRelance; let eventInfo = { id: "", link: "" }; if (rdvDateTime && emailSalarie) { // Un rendez-vous précis a été renseigné : événement à l'horaire exact, créé immédiatement const dateHeureRdv = new Date(rdvDateTime); dateRelance = dateHeureRdv; eventInfo = creerEvenementRendezVous(id, client, typeDemande, salarieAssigne, emailSalarie, dateHeureRdv); } else { dateRelance = new Date(); dateRelance.setDate(dateRelance.getDate() + (joursAvantRelance || CONFIG.JOURS_AVANT_RELANCE)); if (emailSalarie) { eventInfo = creerEvenementDossier(id, client, typeDemande, salarieAssigne, emailSalarie, dateRelance); } } sheet.appendRow([id, client, typeDemande, salarieAssigne, emailSalarie, dateCreation, dateRelance, "Nouveau", 0, "", notes || "", eventInfo.id, clientEmail || "", clientTelephone || "", eventInfo.link, "", canalContact || "", refOrthopmust || "", creePar || "", creePar || "", dateCreation, natureContact || "Client", societe || "", montantAvoir || "", dossierLie || "", referentType || "", referentNom || "", referentContact || "", etablissement || ""]); return id; // utile si le CRM veut garder une référence croisée } /** * Ajoute idB à la liste des dossiers liés de idA (sans doublon). * Utilisé pour créer un lien bidirectionnel entre deux dossiers. */ function ajouterLienDossier(idA, idB) { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === idA) { const ligne = i + 1; const actuels = data[i][24] ? String(data[i][24]).split(",").map(s => s.trim()).filter(Boolean) : []; if (!actuels.includes(idB)) { actuels.push(idB); sheet.getRange(ligne, 25).setValue(actuels.join(",")); } return; } } } // ==================== API WEB APP (pour l'application mobile "Suivi Dossiers") ==================== /** * IMPORTANT — niveau de sécurité : ceci est un jeton partagé simple, pas une * vraie authentification individuelle. Suffisant pour un usage interne à 6 * personnes sur des données non sensibles (client, type de demande, statut). * Ne JAMAIS y faire transiter de données médicales/patient identifiables. * Changez cette valeur avant déploiement, et ne la publiez nulle part en clair. */ const WEB_APP_ACCESS_TOKEN = "CHANGEZ-MOI-" + "0000"; // à remplacer par une valeur unique function jsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /** * Appelé par l'application mobile pour récupérer la liste des dossiers. * URL : {WEB_APP_URL}?token=VOTRE_TOKEN */ function doGet(e) { if (!e || e.parameter.token !== WEB_APP_ACCESS_TOKEN) { return jsonResponse({ error: "Accès refusé" }); } try { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); const dossiers = []; for (let i = 1; i < data.length; i++) { const row = data[i]; if (!row[0]) continue; dossiers.push({ id: row[0], client: row[1], type: row[2], salarie: row[3], email: row[4], dateCreation: row[5], dateRelance: row[6], statut: row[7], nbRelances: row[8], notes: row[10], clientEmail: row[12], clientTelephone: row[13], eventLink: row[14], suiteDonnee: row[15], canal: row[16], refOrthopmust: row[17], creePar: row[18], modifiePar: row[19], dateModification: row[20], nature: row[21] || "Client", societe: row[22], montantAvoir: row[23], dossiersLies: row[24], referentType: row[25], referentNom: row[26], referentContact: row[27], etablissement: row[28] }); } const salariesSheet = getSalariesSheet(); const salariesData = salariesSheet.getDataRange().getValues(); const salaries = []; for (let i = 1; i < salariesData.length; i++) { const row = salariesData[i]; if (!row[0]) continue; salaries.push({ nom: row[0], email: row[1] }); } return jsonResponse({ dossiers: dossiers, salaries: salaries, calendarId: getSharedCalendar().getId() }); } catch (err) { return jsonResponse({ error: "Erreur serveur (doGet) : " + err.message }); } } /** * Appelé par l'application mobile pour ajouter un dossier ou en clôturer un. * Corps de la requête POST attendu (JSON) : * { "token": "...", "action": "add", "client": "...", "type": "...", * "salarie": "...", "email": "...", "notes": "..." } * { "token": "...", "action": "close", "id": "DOS-..." } */ function doPost(e) { let body; try { body = JSON.parse(e.postData.contents); } catch (err) { return jsonResponse({ error: "Requête invalide" }); } if (body.token !== WEB_APP_ACCESS_TOKEN) { return jsonResponse({ error: "Accès refusé" }); } try { if (body.action === "add") { const id = ajouterDossierDepuisCRM(body.client, body.type, body.salarie, body.email, body.notes, undefined, body.clientEmail, body.clientTelephone, body.rdvDateTime, body.canal, body.refOrthopmust, body.creePar, body.nature, body.societe, body.montantAvoir, body.dossierLie, body.referentType, body.referentNom, body.referentContact, body.etablissement); if (body.dossierLie) { ajouterLienDossier(body.dossierLie, id); // lien réciproque sur le dossier d'origine } return jsonResponse({ success: true, id: id }); } if (body.action === "modifier") { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === body.id) { const ligne = i + 1; if (body.client !== undefined) sheet.getRange(ligne, 2).setValue(body.client); if (body.type !== undefined) sheet.getRange(ligne, 3).setValue(body.type); if (body.salarie !== undefined) sheet.getRange(ligne, 4).setValue(body.salarie); if (body.email !== undefined) sheet.getRange(ligne, 5).setValue(body.email); if (body.notes !== undefined) sheet.getRange(ligne, 11).setValue(body.notes); if (body.clientEmail !== undefined) sheet.getRange(ligne, 13).setValue(body.clientEmail); if (body.clientTelephone !== undefined) sheet.getRange(ligne, 14).setValue(body.clientTelephone); if (body.canal !== undefined) sheet.getRange(ligne, 17).setValue(body.canal); if (body.refOrthopmust !== undefined) sheet.getRange(ligne, 18).setValue(body.refOrthopmust); if (body.statut !== undefined) sheet.getRange(ligne, 8).setValue(body.statut); if (body.nature !== undefined) sheet.getRange(ligne, 22).setValue(body.nature); if (body.societe !== undefined) sheet.getRange(ligne, 23).setValue(body.societe); if (body.montantAvoir !== undefined) sheet.getRange(ligne, 24).setValue(body.montantAvoir); if (body.referentType !== undefined) sheet.getRange(ligne, 26).setValue(body.referentType); if (body.referentNom !== undefined) sheet.getRange(ligne, 27).setValue(body.referentNom); if (body.referentContact !== undefined) sheet.getRange(ligne, 28).setValue(body.referentContact); if (body.etablissement !== undefined) sheet.getRange(ligne, 29).setValue(body.etablissement); sheet.getRange(ligne, 20).setValue(body.modifiePar || ""); // Modifié_Par sheet.getRange(ligne, 21).setValue(new Date()); // Date_Modification return jsonResponse({ success: true }); } } return jsonResponse({ error: "Dossier introuvable" }); } if (body.action === "close") { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === body.id) { const ligne = i + 1; if (body.suiteDonnee) { const dateStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm"); const ligneHistorique = `[${dateStr} - ${body.modifiePar || "?"}] ${body.suiteDonnee}`; const historiqueActuel = sheet.getRange(ligne, 16).getValue(); const nouvelHistorique = historiqueActuel ? historiqueActuel + "\n" + ligneHistorique : ligneHistorique; sheet.getRange(ligne, 16).setValue(nouvelHistorique); // colonne P : Suite_Donnée (historique cumulé) } if (body.relancerLe) { // Pas une clôture définitive : le dossier reste ouvert avec une nouvelle date de relance sheet.getRange(ligne, 8).setValue("En cours"); sheet.getRange(ligne, 9).setValue(0); // on repart à 0 relance sheet.getRange(ligne, 7).setValue(new Date(body.relancerLe)); } else { sheet.getRange(ligne, 8).setValue("Clôturé"); } sheet.getRange(ligne, 20).setValue(body.modifiePar || ""); // Modifié_Par sheet.getRange(ligne, 21).setValue(new Date()); // Date_Modification return jsonResponse({ success: true }); } } return jsonResponse({ error: "Dossier introuvable" }); } if (body.action === "addSalarie") { if (!body.nom || !body.email) { return jsonResponse({ error: "Nom et email requis" }); } const salariesSheet = getSalariesSheet(); const existants = salariesSheet.getDataRange().getValues(); const emailNormalise = body.email.trim().toLowerCase(); const dejaPresent = existants.some((row, i) => i > 0 && String(row[1]).trim().toLowerCase() === emailNormalise); if (!dejaPresent) { salariesSheet.appendRow([body.nom, body.email]); } return jsonResponse({ success: true }); } return jsonResponse({ error: "Action inconnue" }); } catch (err) { // Renvoie le détail exact de l'erreur à l'application, pour diagnostic facile return jsonResponse({ error: "Erreur serveur : " + err.message }); } }
